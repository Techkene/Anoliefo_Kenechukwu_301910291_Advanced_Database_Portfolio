-- phase: before

SELECT recorded_at, ph, dissolved_oxygen_mgl, temperature_c, ammonia_mgl
FROM aqua.water_quality_reading
WHERE unit_id = 316
  AND recorded_at >= TIMESTAMPTZ '2026-11-03 23:59+01' - INTERVAL '7 days'
ORDER BY recorded_at DESC
